cmake_minimum_required(VERSION 3.21)
project(ApexOfficeReader VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets)
find_package(Qt6 6.4 QUIET COMPONENTS Pdf PdfWidgets)
add_executable(ApexOfficeReader WIN32 app/main.cpp app/main_window.cpp app/main_window.h core/capability_registry.cpp engines/text_document.cpp)
target_include_directories(ApexOfficeReader PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ApexOfficeReader PRIVATE Qt6::Widgets)
if(TARGET Qt6::Pdf AND TARGET Qt6::PdfWidgets)
 target_link_libraries(ApexOfficeReader PRIVATE Qt6::Pdf Qt6::PdfWidgets)
 target_compile_definitions(ApexOfficeReader PRIVATE APEX_HAS_PDF=1)
endif()
target_compile_definitions(ApexOfficeReader PRIVATE APEX_VERSION="${PROJECT_VERSION}")
if(MSVC)
 target_compile_options(ApexOfficeReader PRIVATE /W4 /permissive-)
else()
 target_compile_options(ApexOfficeReader PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS ApexOfficeReader RUNTIME DESTINATION .)

include(CTest)
if(BUILD_TESTING)
 add_executable(apex_document_tests tests/test_document_basics.cpp core/capability_registry.cpp engines/text_document.cpp)
 target_include_directories(apex_document_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(apex_document_tests PRIVATE Qt6::Core)
 add_test(NAME document_basics COMMAND apex_document_tests)
endif()
